# Dependencies are added ONLY in the task that needs them, and each one is
# justified. No DVC / FastAPI yet -- those arrive later.
#
# Versions are pinned (==) so that this environment can be rebuilt identically
# on another machine, in Docker, or in CI. "It worked yesterday" is not a
# reproducibility strategy.

pandas==2.2.3        # dataframes: the table abstraction for all our tabular data
numpy==1.26.4        # numeric arrays; pandas sits on top of it
pyarrow==18.1.0      # lets pandas read/write Parquet (typed, compressed, fast)
PyYAML==6.0.2        # reads configs/data.yaml so paths are not hard-coded
pytest==8.3.4        # test runner; tests are how "reproducible" becomes real

# Data validation. Pandera lets us declare a DataFrame's contract (columns,
# dtypes, ranges, nullability) as readable code instead of a pile of if/raise
# checks, and reports exactly which rows violated which rule. Chosen over
# Great Expectations, which is far heavier than this project needs.
pandera==0.33.1

# Machine learning. scikit-learn provides the models (linear regression, random
# forest, and a "predict the average" baseline) behind one shared fit/predict
# interface, so swapping models is a one-line change. It also brings joblib,
# which we use to save trained models to disk. Arrives now because this is the
# first task that trains a model.
scikit-learn==1.9.1

# Experiment tracking -- the CLIENT only. The MLflow server runs in Docker
# (compose.yaml); our project just sends it results over the network, so we
# install the "skinny" client (33 packages) rather than full mlflow (70, with
# the web server and database tooling we do not need here).
# Must match the server version in compose.yaml.
mlflow-skinny==3.17.1

# Safe model files. MLflow saves scikit-learn models with skops rather than
# pickle: loading a pickle file can run any code hidden inside it, so a
# tampered model on a shared server could take over the machine that opens it.
# skops only loads known, safe kinds of objects.
skops==0.16.0
